<div class="box">

  <div id="reader"></div>

  <button id="start">開始掃描</button>

  <label>相機無法使用時，改用拍照辨識：

    <input type="file" id="file" accept="image/*" capture="environment"></label>

  <input id="manual" placeholder="或手動輸入登錄號（例如 3000111）">

  <button class="gray" id="manualGo">查詢</button>

</div>


<div class="box" id="card" style="display:none"></div>

<div id="msg" class="box" style="display:none"></div>


<script src="https://unpkg.com/html5-qrcode@2.3.8/html5-qrcode.min.js"></script>

<script>

const API = 'https://script.google.com/macros/s/AKfycbxtdsSc31OX5Yx7cXj17WGbpvD4thaJFqDKVRmyGbg4XA9nDyA-UhoL7cT73b25pcz6/exec';

const $ = id => document.getElementById(id);

let scanner = null, current = null, busy = false;


['who','token'].forEach(k => {

  $(k).value = localStorage.getItem(k) || '';

  $(k).addEventListener('change', () => localStorage.setItem(k, $(k).value.trim()));

});


async function call(payload) {

  const r = await fetch(API, { method: 'POST',

    body: JSON.stringify({ ...payload, token: $('token').value.trim() }) });

  return r.json();

}


function say(t, cls) {

  const m = $('msg'); m.style.display = 'block';

  m.innerHTML = `<span class="${cls||''}">${t}</span>`;

}


async function startScan() {

  if (!scanner) scanner = new Html5Qrcode('reader');

  try {

    await scanner.start({ facingMode: 'environment' },

      { fps: 10, qrbox: 250 }, onCode);

  } catch (e) { say('相機無法啟動，請改用下方拍照辨識或手動輸入', 'warn'); }

}


async function onCode(code) {

  if (busy) return;

  busy = true;

  try { if (scanner && scanner.isScanning) await scanner.stop(); } catch (e) {}

  await lookup(code);

  busy = false;

}


async function lookup(code) {

  if (!$('who').value.trim()) { say('請先輸入盤點人姓名', 'warn'); return; }

  say('查詢中…');

  let res;

  try { res = await call({ action: 'lookup', code, who: $('who').value.trim() }); }

  catch (e) { say('連線失敗，請確認網路後重試', 'warn'); return; }

  if (!res.ok) { say(res.msg, 'warn'); $('card').style.display = 'none'; return; }

  current = { code, items: res.items, item: null };

  $('msg').style.display = 'none';

  if (res.items.length === 1) pick(0); else showChoices(res.items);

}


function pick(i) {

  current.item = current.items[i];

  showCard(current.item);

}


// 撞號：同一個登錄號對應多個品項，讓同仁對照標籤上的物品編號選擇

function showChoices(items) {

  const btns = items.map((it, i) => `

    <button class="gray" data-i="${i}" style="text-align:left">

      <b>${it.物品編號}</b>｜${it.財產名稱}${it.盤點狀態 === '已盤點' ? '（已盤點）' : ''}<br>

      <small>${it.型式} ／ ${it.存放地點} ／ ${it.保管人}</small>

    </button>`).join('');

  $('card').style.display = 'block';

  $('card').innerHTML =

    `<p class="warn">此登錄號對應 ${items.length} 個品項，請對照標籤上的「物品編號」選擇：</p>

     ${btns}<button id="cancel">取消，重新掃描</button>`;

  $('card').querySelectorAll('button[data-i]').forEach(b => b.onclick = () => pick(+b.dataset.i));

  $('cancel').onclick = reset;

}


function showCard(it) {

  const rows = ['物品編號','財產名稱','型式','存放地點','保管人','領用單位','購置日期']

    .map(k => `<tr><td>${k}</td><td>${it[k] ?? ''}</td></tr>`).join('');

  const done = it.盤點狀態 === '已盤點'

    ? `<p class="ok">此品項已於 ${it.盤點時間} 由 ${it.盤點人} 盤點</p>` : '';

  const mine = it.notMine

    ? `<p class="warn">注意：此品項保管人為 ${it.保管人}，非您本人</p>` : '';

  $('card').style.display = 'block';

  $('card').innerHTML = `<table>${rows}</table>${done}${mine}

    <select id="cond"><option>正常</option><option>損壞</option><option>待報廢</option><option>地點不符</option></select>

    <input id="note" placeholder="備註（選填）">

    <button id="ok">確認盤點</button>

    <button class="gray" id="cancel">取消，掃下一個</button>`;

  $('ok').onclick = confirmIt;

  $('cancel').onclick = reset;

}


async function confirmIt() {

  $('ok').disabled = true;

  let res;

  try {

    res = await call({ action: 'confirm', code: current.code,

      itemNo: current.item.物品編號,

      who: $('who').value.trim(), condition: $('cond').value, note: $('note').value.trim() });

  } catch (e) { say('連線失敗，請重試', 'warn'); $('ok').disabled = false; return; }

  if (res.ok) { say('✔ ' + current.item.財產名稱 + ' 盤點完成', 'ok'); reset(true); }

  else { say(res.msg, 'warn'); $('ok').disabled = false; }

}


function reset(keepMsg) {

  $('card').style.display = 'none'; current = null;

  if (keepMsg !== true) $('msg').style.display = 'none';

  startScan();

}


$('start').onclick = startScan;

$('manualGo').onclick = () => { const v = $('manual').value.trim(); if (v) lookup(v); };

$('file').onchange = async e => {

  const f = e.target.files[0]; if (!f) return;

  if (!scanner) scanner = new Html5Qrcode('reader');

  try { const code = await scanner.scanFile(f, false); await lookup(code); }

  catch (err) { say('辨識不到 QR code，請靠近一點再拍', 'warn'); }

  e.target.value = '';

};
